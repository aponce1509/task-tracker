## Issues

```dataviewjs
// 1. Get current active PARA links/paths from the weekly note
const activeParas = dv.current().active_paras || [];
const activePaths = activeParas.map(p => p?.path || p);

// 2. Custom ordering weights
const statusOrder = { "inbox": 1, "backlog": 2, "wip": 3, "blocked": 4 };
const getStatusWeight = (s) => statusOrder[s] || 99;

// 3. Filter issues matching active_paras, excluding 'done'
const issues = dv.pages()
    .where(p => p.file_class === "issue" || (p.tags && p.tags.includes("issue")))
    .where(p => p.status !== "done")
    .where(p => {
        if (!p.paras) return false;
        const pageParas = Array.isArray(p.paras) ? p.paras : [p.paras];
        return pageParas.some(para => activePaths.includes(para?.path || para));
    });

// 4. Group by PARA -> Status -> Priority
const grouped = issues.groupBy(p => {
    // Find which active PARA this issue belongs to
    const pageParas = Array.isArray(p.paras) ? p.paras : [p.paras];
    const matchedPara = pageParas.find(para => activePaths.includes(para?.path || para));
    return matchedPara?.path || matchedPara || "Unassigned";
});

// 5. Render Hierarchy
for (let paraGroup of grouped) {
    dv.header(2, `📁 PARA: ${paraGroup.key.replace(/\.md$/, "")}`);

    // Group & Sort by Status
    const statusGroups = paraGroup.rows.groupBy(p => {
        const rawStatus = (p.status || "").trim().toLowerCase();
        return rawStatus === "" ? "backlog" : rawStatus;
    }).sort(g => getStatusWeight(g.key), "asc");

    for (let statusGroup of statusGroups) {
        dv.header(3, `Status: ${statusGroup.key.toUpperCase()}`);

        // Group & Sort by Priority (0, 1, 2...)
        const prioGroups = statusGroup.rows.groupBy(p => p.priority ?? "No Prio")
            .sort(g => g.key, "asc");

        for (let prioGroup of prioGroups) {
            dv.header(4, `Priority ${prioGroup.key}`);

            // Sort issues within group by dueDate (earliest first, nulls last)
            const sortedIssues = prioGroup.rows.sort(p => p.dueDate || "9999-99-99", "asc");

            dv.table(["Issue", "Due Date"],
                sortedIssues.map(p => [
                    p.file.link,
                    p.dueDate || "—",
                ])
            );
        }
    }
}
```

## Tags Search

### Whitelist

```dataviewjs
// 1. Tag Whitelist: ONLY notes with these tags will be included (include '#' prefix)
const allowedTags = ["#todo", "#extract", "#debt", "#verify", "#missing"];

// 2. Normalize allowed tags list
const cleanTag = (t) => t.startsWith("#") ? t : `#${t}`;
const whiteList = allowedTags.map(cleanTag);

// 3. Gather all non-'done' pages
const pages = dv.pages();

// 4. Flatten notes by whitelisted tags (multi-tag notes appear in each matched group)
const taggedEntries = [];
for (let page of pages) {
    const rawTags = page.file.tags || [];
    const matchedWhitelist = rawTags.filter(t => whiteList.includes(t));
    
    for (let tag of matchedWhitelist) {
        taggedEntries.push({ tag: tag, page: page });
    }
}

// 5. Group by individual Whitelisted Tag
const tagGroups = dv.array(taggedEntries)
    .groupBy(e => e.tag)
    .sort(g => g.key, "asc");

// 6. Render Hierarchy: Tag -> Priority -> Due Date
for (let tagGroup of tagGroups) {
    dv.header(2, `Tag: ${tagGroup.key}`);

	// Sort individual notes inside group by dueDate (earliest first, nulls last)
	const sortedEntries = tagGroup.rows;

	dv.table(["Note", "Status", "Folder"],
		sortedEntries.map(e => [
			e.page.file.link,
			(!e.page.status || e.page.status.trim() === "") ? "backlog" : e.page.status,
			e.page.file.folder,
		])
	);
}
```
### Blacklisted

```dataviewjs
// 1. Tags to filter out / ignore (include '#' prefix)
const excludedTags = [];

// 2. Normalize excluded tags list
const cleanTag = (t) => t.startsWith("#") ? t : `#${t}`;
const ignoreList = excludedTags.map(cleanTag);

// 3. Gather all pages and exclude 'done' items
const pages = dv.pages();

// 4. Flatten notes by tag so multi-tag notes appear in each tag group
const taggedEntries = [];
for (let page of pages) {
    const rawTags = page.file.tags || [];
    const validTags = rawTags.filter(t => !ignoreList.includes(t));
    
    if (validTags.length === 0) {
        taggedEntries.push({ tag: "#untagged", page: page });
    } else {
        for (let tag of validTags) {
            taggedEntries.push({ tag: tag, page: page });
        }
    }
}

// 5. Group by individual Tag
const tagGroups = dv.array(taggedEntries)
    .groupBy(e => e.tag)
    .sort(g => g.key, "asc");

// 6. Render Hierarchy: Tag -> Priority -> Due Date
for (let tagGroup of tagGroups) {
    dv.header(2, `Tag: ${tagGroup.key}`);

	// Sort individual notes inside group by dueDate (earliest first, nulls last)
	const sortedEntries = tagGroup.rows;

	dv.table(["Note", "Status", "Folder"],
		sortedEntries.map(e => [
			e.page.file.link,
			(!e.page.status || e.page.status.trim() === "") ? "backlog" : e.page.status,
			e.page.file.folder,
		])
	);
}
```
